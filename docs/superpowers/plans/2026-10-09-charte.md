# Charte « Nuit mauve » — plan d'implémentation

> **Découpé en fiches** : exécuter le plan à partir de `charte/LISEZMOI.md` (une conversation par fiche, sorties courtes, relectures groupées). Ce fichier complet reste la référence d'historique.

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Repeindre tout le site aux couleurs de la charte sombre « Nuit mauve » (sauge, mauve, rose, corail, pêche sur un fond nuit), avec des jetons à deux étages, une intro « goutte de lumière » et des garde-fous de contraste.

**Architecture:** Les teintes brutes sont des variables CSS `--teinte-*` sur `:root` ; les rôles (`fond`, `texte`, `accent`…) sont les seuls jetons `@theme` de Tailwind (palette par défaut retirée), donc les seules classes de couleur possibles. `lib/charte.ts` porte les mêmes valeurs pour le code TypeScript (intro WebGL, vignettes, curseurs, scène 3D) ; des tests tiennent CSS et TypeScript d'accord, vérifient chaque contraste et refusent toute classe de couleur hors des rôles. La démo de l'étude de cas bascule en sombre par `?theme=sombre`.

**Tech Stack:** Next.js 16.4 (App Router), React 19.2, Tailwind CSS 4.3, TypeScript, Vitest (jsdom), Playwright, `@axe-core/playwright` (nouveau), Vite (démo), Hono (service temps réel), Three.js (scène 3D des Coulisses).

**Spec:** `docs/superpowers/specs/2026-10-09-charte-design.md` (à lire avant toute tâche).

**Où exécuter :** la spec et ce plan n'existent que sur la branche `sous-projet-6-charte`, dans le worktree `.superpowers/worktrees/charte` du dépôt. Toutes les commandes se lancent depuis la racine de ce worktree, jamais depuis le dossier principal du dépôt, où d'autres travaux peuvent être en cours.

```ts
/**
 * Charte « Nuit mauve » pour le code qui ne lit pas le CSS : moteur de l'intro,
 * vignettes de partage, curseurs du direct, scène 3D des Coulisses. Mêmes
 * valeurs que app/globals.css (`--teinte-*` sur `:root`, rôles dans `@theme`),
 * ce que vérifie lib/charte.test.ts.
 *
 * Les composants n'emploient que les rôles ; les teintes brutes ne servent qu'à
 * les définir.
 */
export const TEINTES = {
  nuit: "#2a2127",
  "nuit-haute": "#352a31",
  mauve: "#79616f",
  "mauve-clair": "#b9a3b0",
  rose: "#ae6378",
  corail: "#d87f81",
  peche: "#eab595",
  sauge: "#7e9680",
  papier: "#f3efe7",
  "trait-nuit": "#4a3b44",
  "trait-mauve": "#8a7480",
} as const;

export type Teinte = keyof typeof TEINTES;

export const ROLES = {
  fond: "nuit",
  surface: "nuit-haute",
  texte: "papier",
  "texte-doux": "mauve-clair",
  accent: "peche",
  "sur-accent": "nuit",
  alerte: "corail",
  vivant: "sauge",
  trait: "trait-nuit",
  "trait-fort": "trait-mauve",
  lueur: "corail",
  "lueur-chaude": "rose",
  goutte: "peche",
} as const satisfies Record<string, Teinte>;

export type Role = keyof typeof ROLES;

/** Couleur d'un rôle, en `#rrggbb` minuscule. */
export function couleur(role: Role): string {
  return TEINTES[ROLES[role]];
}

/** Ratio de contraste WCAG 2.x entre deux couleurs `#rrggbb` (de 1 à 21). */
export function contraste(a: string, b: string): number {
  const luminance = (hex: string) => {
    const [r, v, bl] = [1, 3, 5].map((i) => {
      const c = parseInt(hex.slice(i, i + 2), 16) / 255;
      return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
    });
    return 0.2126 * r! + 0.7152 * v! + 0.0722 * bl!;
  };
  const [claire, sombre] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (claire! + 0.05) / (sombre! + 0.05);
}

/**
 * Contrastes exigés : 4,5:1 pour le texte (WCAG 1.4.3), 3:1 pour les contours de
 * champs et de boutons, les pastilles et l'anneau de focus (1.4.11). La sauge
 * (`vivant`) n'est jamais du texte : sur une carte, elle n'atteint que 4,3:1.
 * Le nom de l'intro se lit en `sur-accent` sur la `goutte`.
 */
export const CONTRASTES_EXIGES: readonly { avant: Role; arriere: Role; minimum: number }[] = [
  { avant: "texte", arriere: "fond", minimum: 4.5 },
  { avant: "texte", arriere: "surface", minimum: 4.5 },
  { avant: "texte-doux", arriere: "fond", minimum: 4.5 },
  { avant: "texte-doux", arriere: "surface", minimum: 4.5 },
  { avant: "accent", arriere: "fond", minimum: 4.5 },
  { avant: "accent", arriere: "surface", minimum: 4.5 },
  { avant: "sur-accent", arriere: "accent", minimum: 4.5 },
  { avant: "sur-accent", arriere: "goutte", minimum: 4.5 },
  { avant: "alerte", arriere: "fond", minimum: 4.5 },
  { avant: "alerte", arriere: "surface", minimum: 4.5 },
  { avant: "vivant", arriere: "fond", minimum: 3 },
  { avant: "vivant", arriere: "surface", minimum: 3 },
  { avant: "trait-fort", arriere: "fond", minimum: 3 },
  { avant: "trait-fort", arriere: "surface", minimum: 3 },
];
```

```ts
import { readdirSync, readFileSync, statSync } from "node:fs";
import { join } from "node:path";
import { fileURLToPath } from "node:url";
import { describe, expect, it } from "vitest";
import { TEINTES } from "./charte";

/**
 * Les composants n'emploient que les rôles de la charte, sans opacité. Tailwind 4
 * ignore en silence une classe dont la couleur n'existe pas : sans ce test, une
 * ancienne classe donnerait un texte sans couleur sur la nuit, sans erreur.
 */
const ANCIENS = ["papier", "encre", "encre-douce", "brique", "braise", "braise-claire", "terre"];
const PALETTE_TAILWIND = [
  "white", "black", "slate", "gray", "zinc", "neutral", "stone", "red", "orange", "amber", "yellow",
  "lime", "green", "emerald", "teal", "cyan", "sky", "blue", "indigo", "violet", "purple", "fuchsia",
  "pink", "rose",
];
const INTERDITES = [...new Set([...ANCIENS, ...Object.keys(TEINTES), ...PALETTE_TAILWIND])]
  // Les noms longs d'abord : `encre-douce` avant `encre`.
  .sort((a, b) => b.length - a.length);
const PREFIXES =
  "bg|text|border(?:-[trblxy])?|ring(?:-offset)?|outline|fill|stroke|decoration|divide|from|via|to|shadow|accent|caret|placeholder";
const CLASSE_INTERDITE = new RegExp(
  `(?<![\\w-])(?:[\\w-]+:)*(?:${PREFIXES})-(?:${INTERDITES.join("|")})(?:-\\d{2,3})?(?:/\\d+)?(?![\\w-])`,
  "g",
);
/** Couleur avec opacité (`text-texte/75`) : le contraste n'est plus garanti. */
const OPACITE = /(?<![\w-])(?:[\w-]+:)*(?:bg|text|border|ring|fill|stroke|decoration)-[a-z-]+\/\d+(?![\w-])/g;
/** La sauge n'est jamais du texte : 4,3:1 seulement sur une carte. */
const TEXTE_VIVANT = /(?<![\w-])(?:[\w-]+:)*text-vivant(?![\w-])/g;
/** Variable d'un ancien jeton (`stroke="var(--color-papier)"`). */
const VARIABLE_ANCIENNE = new RegExp(`var\\(--color-(?:${ANCIENS.join("|")})\\)`, "g");

export function classesInterdites(source: string): string[] {
  return [CLASSE_INTERDITE, OPACITE, TEXTE_VIVANT, VARIABLE_ANCIENNE].flatMap((motif) => source.match(motif) ?? []);
}

const racine = fileURLToPath(new URL("..", import.meta.url));
function fichiers(dossier: string): string[] {
  return readdirSync(dossier).flatMap((nom) => {
    const chemin = join(dossier, nom);
    if (statSync(chemin).isDirectory()) return fichiers(chemin);
    return /\.tsx?$/.test(nom) && !chemin.endsWith("charte-classes.test.ts") ? [chemin] : [];
  });
}

describe("classesInterdites", () => {
  it("repère anciens jetons, teintes brutes, palette Tailwind, opacités, text-vivant et anciennes variables", () => {
    expect(
      classesInterdites(
        'className="bg-emerald-600 text-encre-douce hover:text-encre bg-peche ring-offset-papier text-texte/75 text-vivant" stroke="var(--color-papier)"',
      ),
    ).toEqual([

      "text-encre-douce",
      "hover:text-encre",
      "bg-peche",
      "ring-offset-papier",
      "text-texte/75",
      "text-vivant",
      "var(--color-papier)",
    ]);
  });

  it("laisse passer les rôles et les utilitaires sans couleur", () => {
    expect(
      classesInterdites(
        'className="bg-fond text-texte-doux border-trait-fort ring-accent ring-offset-fond text-sm border-t ring-2 text-sur-accent bg-vivant decoration-accent fill-surface stroke-trait" stroke="var(--color-fond)"',
      ),
    ).toEqual([]);
  });
});

describe("app/, components/, lib/", () => {
  it("n'emploient que des rôles de la charte", () => {
    const fautes = ["app", "components", "lib"]
      .flatMap((d) => fichiers(join(racine, d)))
      .flatMap((f) => classesInterdites(readFileSync(f, "utf8")).map((c) => `${f.slice(racine.length)} : ${c}`));
    expect(fautes).toEqual([]);
  });
});
```

```ts
import AxeBuilder from "@axe-core/playwright";
import { expect, test, type Page } from "@playwright/test";
import { simulerGpu } from "./outils";

/**
 * Charte « Nuit mauve » : chaque page publique passe les règles WCAG 2.0, 2.1 et
 * 2.2 (A et AA) d'axe-core, contraste compris : plus large que Lighthouse.
 */
const NORMES = ["wcag2a", "wcag2aa", "wcag21a", "wcag21aa", "wcag22aa"];
const PAGES = [
  "/",
  "/realisations/suivi-de-marches",
  "/realisations/site-artisan",
  "/contact",
  "/coulisses",
  "/mentions-legales",
  "/politique-de-confidentialite",
  "/admin/connexion",
  "/demo/suivi-de-marches/index.html?theme=sombre",
];

async function violations(page: Page): Promise<string[]> {
  const resultat = await new AxeBuilder({ page }).withTags(NORMES).analyze();
  return resultat.violations.map((v) => `${v.id} : ${v.nodes.map((n) => n.target.join(" ")).join(", ")}`);
}

test.describe("mouvement réduit (page posée)", () => {
  test.use({ reducedMotion: "reduce" });

  for (const chemin of PAGES) {
    test(`${chemin} : aucune violation WCAG A/AA`, async ({ page }) => {
      await page.goto(chemin);
      expect(await violations(page)).toEqual([]);
    });
  }

  test("page personnalisée d'exemple : aucune violation WCAG A/AA", async ({ page }) => {
    test.skip(!process.env.DATABASE_URL, "Base de test requise (DATABASE_URL).");
    await page.goto("/?pour=exemple-0000");
    expect(await violations(page)).toEqual([]);
  });
});

test("accueil après l'intro, sans mouvement réduit : aucune violation WCAG A/AA", async ({ page }) => {
  await simulerGpu(page);
  await page.goto("/");
  await expect(page.locator(".fond-accueil")).toHaveAttribute("data-intro", "terminee", { timeout: 5000 });
  expect(await violations(page)).toEqual([]);
});

test("la page déclare un thème sombre aux contrôles natifs", async ({ page }) => {
  await page.goto("/");
  expect(await page.evaluate(() => getComputedStyle(document.documentElement).colorScheme)).toBe("dark");
});

for (const chemin of ["/", "/contact"]) {
  test(`${chemin} : chaque élément atteint au clavier montre un focus pêche`, async ({ page, isMobile }) => {
    test.skip(isMobile, "Navigation au clavier : ordinateur seulement.");
    await page.emulateMedia({ reducedMotion: "reduce" });
    await page.goto(chemin);
    const vus = new Set<string>();
    for (let i = 0; i < 80; i++) {
      await page.keyboard.press("Tab");
      const etat = await page.evaluate(() => {
        const el = document.activeElement as HTMLElement | null;
        if (!el || el === document.body) return null;
        const s = getComputedStyle(el);
        const peche = "rgb(234, 181, 149)";
        const contour = s.outlineStyle !== "none" && parseFloat(s.outlineWidth) >= 2 && s.outlineColor === peche;
        const anneau = s.boxShadow.includes(peche);
        return { id: el.outerHTML.slice(0, 160), visible: contour || anneau };
      });
      if (!etat || vus.has(etat.id)) break;
      vus.add(etat.id);
      expect(etat.visible, etat.id).toBe(true);
    }
    expect(vus.size).toBeGreaterThan(5);
  });
}
```
